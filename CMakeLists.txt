# --- third_party/MANIFEST.toml -> prism/manifest_pins.hpp ---
# The C++ engine trusts ~/.prism/tools/<name>/<commit>/ only for the commit its
# own manifest pins. `prism-deps pins` writes the header with the same
# fail-closed manifest reader the fetcher uses (a malformed manifest stops
# the build), and rewrites it only when the pins change.
set(_pins_dir "${CMAKE_BINARY_DIR}/generated")
set(_pins_hdr "${_pins_dir}/prism/manifest_pins.hpp")
add_custom_command(
  OUTPUT "${_pins_hdr}"
  COMMAND prism-deps pins --manifest "${_prism_manifest}" -o "${_pins_hdr}"
  DEPENDS prism-deps "${_prism_manifest}"
  COMMENT "prism-deps pins: third_party/MANIFEST.toml -> prism/manifest_pins.hpp"
  VERBATIM)

# --- docs shipped with every report -> prism/shipped_docs.inc ---
# docs/TRUSTED_BASE.md (roadmap 3.2 / 8.3) and docs/VERDICTS.md (6.4) are
# written next to report.md on every run (src/prism/shipdocs.cpp,
# prism/shipdocs.py reads the same files). Byte-identical copies.
set(_ship_inc "// Generated by CMake from docs/*.md. Do not edit.\n")
foreach(_doc TRUSTED_BASE VERDICTS)
  set(_doc_path "${PRISM_ROOT}/docs/${_doc}.md")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_doc_path}")
  file(READ "${_doc_path}" _doc_text)
  string(APPEND _ship_inc "inline constexpr std::string_view kDoc_${_doc} = R\"PRISMDOC(${_doc_text})PRISMDOC\";\n")
endforeach()
file(WRITE "${_pins_dir}/prism/shipped_docs.inc.tmp" "${_ship_inc}")
configure_file("${_pins_dir}/prism/shipped_docs.inc.tmp" "${_pins_dir}/prism/shipped_docs.inc" COPYONLY)

# --- grammars/*.gbnf -> prism/ai/grammars.inc ---
# The GBNF grammars the model backends are constrained by
# (src/prism/ai/core.cpp grammar_text), embedded verbatim as raw strings.
# Generated here so the embedded copy cannot drift from grammars/.
set(_gbnf_inc "// Generated by CMake from grammars/*.gbnf. Do not edit.\n")
foreach(_g invariants harness contract explain lean_proof assumption_audit ask draft)
  set(_g_path "${PRISM_ROOT}/grammars/${_g}.gbnf")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_g_path}")
  file(READ "${_g_path}" _g_text)
  string(REPLACE "\r" "" _g_text "${_g_text}")  # autocrlf checkouts
  string(FIND "${_g_text}" ")GBNF\"" _g_bad)
  if(NOT _g_bad EQUAL -1)
    message(FATAL_ERROR "grammars/${_g}.gbnf contains the raw-string delimiter )GBNF\"")
  endif()
  string(TOUPPER "${_g}" _g_up)
  string(APPEND _gbnf_inc "constexpr const char* GBNF_${_g_up} = R\"GBNF(${_g_text})GBNF\";\n")
endforeach()
file(WRITE "${_pins_dir}/prism/ai/grammars.inc.tmp" "${_gbnf_inc}")
configure_file("${_pins_dir}/prism/ai/grammars.inc.tmp" "${_pins_dir}/prism/ai/grammars.inc" COPYONLY)

set(_pir_models_cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/pir_models_gen.cpp")
set(_pir_models_txt "// generated by CMakeLists.txt from src/prism/pir/models (do not edit)\n#include <string>\n#include <utility>\n#include <vector>\nnamespace prism::pir {\nconst std::vector<std::pair<std::string, std::string>>& embedded_models() {\n  static const std::vector<std::pair<std::string, std::string>> k{\n")
foreach(_m ${PRISM_PIR_MODELS})
  file(READ "${PRISM_ROOT}/${_m}" _mtxt)
  get_filename_component(_mn "${_m}" NAME)
  if(_m MATCHES "/models/cxx/")
    # keep the path below models/cxx (bits/basic_string.tcc replaces the
    # libstdc++ file of that name on the include path)
    string(REGEX REPLACE "^.*/models/cxx/" "" _mn "${_m}")
    set(_mn "cxx/${_mn}")
  endif()
  string(APPEND _pir_models_txt "    {\"${_mn}\", R\"PRISMMODEL(${_mtxt})PRISMMODEL\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PRISM_ROOT}/${_m}")
endforeach()
string(APPEND _pir_models_txt "  };\n  return k;\n}\n}  // namespace prism::pir\n")
file(CONFIGURE OUTPUT "${_pir_models_cpp}" CONTENT "${_pir_models_txt}" @ONLY)
list(APPEND PRISM_SRC "${_pir_models_cpp}")

# Roadmap 2.8: the Clang-AST lints parse in process through libclang's C API.
# Only the header is needed at build time: the library is dlopen()ed at run
# time (src/prism/astlint_libclang.cpp), so a machine without libclang.so still
# runs PRISM and the layer falls back to clang as a process. Default ON when
# <clang-c/Index.h> is found (apt install libclang-18-dev).
option(PRISM_LIBCLANG "Clang-AST lints through libclang (header needed at build time)" ON)
if(PRISM_LIBCLANG)
  set(_llvm_hints)
  foreach(_v 21 20 19 18 17)
    list(APPEND _llvm_hints "/usr/lib/llvm-${_v}")
  endforeach()
  list(APPEND _llvm_hints "/usr/local/opt/llvm" "/opt/homebrew/opt/llvm" "C:/Program Files/LLVM")
  find_path(PRISM_CLANG_C_INCLUDE clang-c/Index.h
    HINTS ${_llvm_hints} PATH_SUFFIXES include)
  find_library(PRISM_LIBCLANG_LIBRARY NAMES clang libclang clang-18 clang-19 clang-20 clang-17
    HINTS ${_llvm_hints} PATH_SUFFIXES lib)
  if(PRISM_CLANG_C_INCLUDE)
    target_include_directories(prism_core SYSTEM PRIVATE "${PRISM_CLANG_C_INCLUDE}")
    target_compile_definitions(prism_core PRIVATE PRISM_HAS_LIBCLANG)
    if(PRISM_LIBCLANG_LIBRARY)
      target_compile_definitions(prism_core PRIVATE PRISM_LIBCLANG_HINT="${PRISM_LIBCLANG_LIBRARY}")
    endif()
    message(STATUS "libclang: ${PRISM_CLANG_C_INCLUDE} (run time: ${PRISM_LIBCLANG_LIBRARY}, dlopen)")
  else()
    message(STATUS "libclang header not found: the Clang-AST lints use clang as a process")
  endif()
endif()

  # -allow-unsupported-compiler (that would silently disable the check).
  set(_prism_cuda_host "${CMAKE_CUDA_HOST_COMPILER}")
  if(NOT _prism_cuda_host)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_C_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_CXX_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND UNIX)
      # Prefer gcc-12: CUDA 12.0 nvcc rejects gcc 13 unless given
      # -allow-unsupported-compiler, which would silently disable the check.
      find_program(_PRISM_CUDA_GCC NAMES gcc-12 gcc-11 gcc-10 gcc)
      if(_PRISM_CUDA_GCC)
        set(_prism_cuda_host "${_PRISM_CUDA_GCC}")
      else()
        find_program(_PRISM_CUDA_GXX NAMES g++-12 g++-11 g++-10 g++)
        if(_PRISM_CUDA_GXX)
          set(_prism_cuda_host "${_PRISM_CUDA_GXX}")
        endif()
      endif()
    endif()
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    if(_prism_cuda_host)
      set(CMAKE_CUDA_HOST_COMPILER "${_prism_cuda_host}" CACHE FILEPATH
          "Host compiler for nvcc (gcc, not clang++)" FORCE)
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR _prism_cuda_host)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
      # nvcc 12 has no C++23; do not inherit CMAKE_CXX_STANDARD 23 (clang++/prism_core).
      set(CMAKE_CUDA_STANDARD 17)
      set(CMAKE_CUDA_STANDARD_REQUIRED ON)
      enable_language(CUDA)
      set(_prism_cuda_lang ON)
      if(NOT CMAKE_CUDA_ARCHITECTURES)
        # PLAN stack is an RTX 3090 (sm_86). native needs a live GPU at configure.
        set(CMAKE_CUDA_ARCHITECTURES "86" CACHE STRING "nvcc -arch" FORCE)
      endif()
      # probsat.cu: GPU ProbSAT walkers (roadmap 3.3), mirrors src/prism/solver/cnf.cpp.
      add_library(prism_cuda SHARED src/cuda/mutate.cu src/cuda/probsat.cu)
      target_include_directories(prism_cuda PUBLIC "${PRISM_ROOT}/include")
      target_compile_features(prism_cuda PUBLIC cuda_std_17)
      set_target_properties(prism_cuda PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        BUILD_RPATH "$ORIGIN"
        INSTALL_RPATH "$ORIGIN")
      # prism / prism_tests / prism_gui link prism_core, not prism_native.
      # Leaving the kernel only on prism_native was a silent disable.
      target_link_libraries(prism_core PUBLIC prism_cuda)
      target_link_libraries(prism_native PUBLIC prism_cuda)
      target_compile_definitions(prism_native PUBLIC PRISM_HAS_CUDA)
      target_compile_definitions(prism_core PUBLIC PRISM_HAS_CUDA)
      message(STATUS "CUDA mutation kernel ON (nvcc host ${CMAKE_CUDA_HOST_COMPILER})")
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
else()
  message(WARNING "PRISM_CUDA=OFF — CUDA kernel skipped (NOTRUN, not a silent disable)")
endif()
