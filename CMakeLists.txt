# third_party/ holds only the six linked libraries (xsimd, z3, pcre2, nlohmann,
# doctest, llama.cpp), each pinned in third_party/MANIFEST.toml. External tools
# are fetched by scripts/fetch_deps.py into ~/.prism/tools/, never built here.
# Never file(GLOB) third_party/.

# --- third_party/MANIFEST.toml -> prism/manifest_pins.hpp ---
# The C++ engine trusts ~/.prism/tools/<name>/<commit>/ only for the commit its
# own manifest pins (prism/config.py reads the same file with tomllib).
set(_prism_manifest "${PRISM_ROOT}/third_party/MANIFEST.toml")
if(NOT EXISTS "${_prism_manifest}")
  message(FATAL_ERROR "third_party/MANIFEST.toml missing")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_prism_manifest}")
file(READ "${_prism_manifest}" _mf)
# Neutralise list syntax before splitting into lines: [ ] ; are CMake list metacharacters.
string(REPLACE "[" "<" _mf "${_mf}")
string(REPLACE "]" ">" _mf "${_mf}")
string(REPLACE ";" "," _mf "${_mf}")
string(REPLACE "\n" ";" _mf_lines "${_mf}")
set(_pins "")
set(_n "")
set(_k "")
set(_c "")
foreach(_line IN LISTS _mf_lines "<<end>>")
  if(_line MATCHES "^<<")
    if(_k STREQUAL "external" AND _n AND _c)
      string(APPEND _pins "    {\"${_n}\", \"${_c}\"},\n")
    endif()
    set(_n "")
    set(_k "")
    set(_c "")
  elseif(_line MATCHES "^name = \"([^\"]+)\"")
    set(_n "${CMAKE_MATCH_1}")
  elseif(_line MATCHES "^kind = \"([^\"]+)\"")
    set(_k "${CMAKE_MATCH_1}")
  elseif(_line MATCHES "^commit = \"([0-9a-f]+)\"")
    set(_c "${CMAKE_MATCH_1}")
  endif()
endforeach()
set(_pins_dir "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${_pins_dir}/prism/manifest_pins.hpp.tmp"
"// Generated by CMake from third_party/MANIFEST.toml. Do not edit.\n"
"#pragma once\n\nnamespace prism {\n"
"struct ManifestPin {\n    const char* name;\n    const char* commit;\n};\n"
"inline constexpr ManifestPin kManifestPins[] = {\n${_pins}};\n}  // namespace prism\n")
configure_file("${_pins_dir}/prism/manifest_pins.hpp.tmp" "${_pins_dir}/prism/manifest_pins.hpp" COPYONLY)

set(_pir_models_cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/pir_models_gen.cpp")
set(_pir_models_txt "// generated by CMakeLists.txt from src/prism/pir/models (do not edit)\n#include <string>\n#include <utility>\n#include <vector>\nnamespace prism::pir {\nconst std::vector<std::pair<std::string, std::string>>& embedded_models() {\n  static const std::vector<std::pair<std::string, std::string>> k{\n")
foreach(_m ${PRISM_PIR_MODELS})
  file(READ "${PRISM_ROOT}/${_m}" _mtxt)
  get_filename_component(_mn "${_m}" NAME)
  string(APPEND _pir_models_txt "    {\"${_mn}\", R\"PRISMMODEL(${_mtxt})PRISMMODEL\"},\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PRISM_ROOT}/${_m}")
endforeach()
string(APPEND _pir_models_txt "  };\n  return k;\n}\n}  // namespace prism::pir\n")
file(CONFIGURE OUTPUT "${_pir_models_cpp}" CONTENT "${_pir_models_txt}" @ONLY)
list(APPEND PRISM_SRC "${_pir_models_cpp}")

set(_prism_cuda_lang OFF)
if(PRISM_CUDA)
  include(CheckLanguage)
  # nvcc cannot use clang++ as host. Keep CMAKE_CXX_COMPILER (clang++) for
  # prism_core; point CMAKE_CUDA_HOST_COMPILER at gcc/g++ *before*
  # check_language / enable_language so nvcc is allowed. Never pass
  # -allow-unsupported-compiler (that would silently disable the check).
  set(_prism_cuda_host "${CMAKE_CUDA_HOST_COMPILER}")
  if(NOT _prism_cuda_host)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_C_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_CXX_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND UNIX)
      # Prefer gcc-12: CUDA 12.0 nvcc rejects gcc 13 unless given
      # -allow-unsupported-compiler, which would silently disable the check.
      find_program(_PRISM_CUDA_GCC NAMES gcc-12 gcc-11 gcc-10 gcc)
      if(_PRISM_CUDA_GCC)
        set(_prism_cuda_host "${_PRISM_CUDA_GCC}")
      else()
        find_program(_PRISM_CUDA_GXX NAMES g++-12 g++-11 g++-10 g++)
        if(_PRISM_CUDA_GXX)
          set(_prism_cuda_host "${_PRISM_CUDA_GXX}")
        endif()
      endif()
    endif()
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    if(_prism_cuda_host)
      set(CMAKE_CUDA_HOST_COMPILER "${_prism_cuda_host}" CACHE FILEPATH
          "Host compiler for nvcc (gcc, not clang++)" FORCE)
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR _prism_cuda_host)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
      # nvcc 12 has no C++23; do not inherit CMAKE_CXX_STANDARD 23 (clang++/prism_core).
      set(CMAKE_CUDA_STANDARD 17)
      set(CMAKE_CUDA_STANDARD_REQUIRED ON)
      enable_language(CUDA)
      set(_prism_cuda_lang ON)
      if(NOT CMAKE_CUDA_ARCHITECTURES)
        # PLAN stack is an RTX 3090 (sm_86). native needs a live GPU at configure.
        set(CMAKE_CUDA_ARCHITECTURES "86" CACHE STRING "nvcc -arch" FORCE)
      endif()
      # probsat.cu: GPU ProbSAT walkers (roadmap 3.3), mirrors src/prism/solver/cnf.cpp.
      add_library(prism_cuda SHARED src/cuda/mutate.cu src/cuda/probsat.cu)
      target_include_directories(prism_cuda PUBLIC "${PRISM_ROOT}/include")
      target_compile_features(prism_cuda PUBLIC cuda_std_17)
      set_target_properties(prism_cuda PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        BUILD_RPATH "$ORIGIN"
        INSTALL_RPATH "$ORIGIN")
      # prism / prism_tests / prism_gui link prism_core, not prism_native.
      # Leaving the kernel only on prism_native was a silent disable.
      target_link_libraries(prism_core PUBLIC prism_cuda)
      target_link_libraries(prism_native PUBLIC prism_cuda)
      target_compile_definitions(prism_native PUBLIC PRISM_HAS_CUDA)
      target_compile_definitions(prism_core PUBLIC PRISM_HAS_CUDA)
      message(STATUS "CUDA mutation kernel ON (nvcc host ${CMAKE_CUDA_HOST_COMPILER})")
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
else()
  message(WARNING "PRISM_CUDA=OFF — CUDA kernel skipped (NOTRUN, not a silent disable)")
endif()
