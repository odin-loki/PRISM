# doctest, llama.cpp), each pinned in third_party/MANIFEST.toml. External tools
# are fetched by scripts/fetch_deps.py into ~/.prism/tools/, never built here.

# --- third_party/MANIFEST.toml -> prism/manifest_pins.hpp ---
# The C++ engine trusts ~/.prism/tools/<name>/<commit>/ only for the commit its
# own manifest pins (prism/config.py reads the same file with tomllib).
set(_prism_manifest "${PRISM_ROOT}/third_party/MANIFEST.toml")
if(NOT EXISTS "${_prism_manifest}")
  message(FATAL_ERROR "third_party/MANIFEST.toml missing")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_prism_manifest}")
file(READ "${_prism_manifest}" _mf)
# Neutralise list syntax before splitting into lines: [ ] ; are CMake list metacharacters.
string(REPLACE "[" "<" _mf "${_mf}")
string(REPLACE "]" ">" _mf "${_mf}")
string(REPLACE ";" "," _mf "${_mf}")
string(REPLACE "\n" ";" _mf_lines "${_mf}")
set(_pins "")
set(_n "")
set(_k "")
set(_c "")
foreach(_line IN LISTS _mf_lines "<<end>>")
  if(_line MATCHES "^<<")
    if(_k STREQUAL "external" AND _n AND _c)
      string(APPEND _pins "    {\"${_n}\", \"${_c}\"},\n")
    endif()
    set(_n "")
    set(_k "")
    set(_c "")
  elseif(_line MATCHES "^name = \"([^\"]+)\"")
    set(_n "${CMAKE_MATCH_1}")
  elseif(_line MATCHES "^kind = \"([^\"]+)\"")
    set(_k "${CMAKE_MATCH_1}")
  elseif(_line MATCHES "^commit = \"([0-9a-f]+)\"")
    set(_c "${CMAKE_MATCH_1}")
  endif()
endforeach()
set(_pins_dir "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${_pins_dir}/prism/manifest_pins.hpp.tmp"
"// Generated by CMake from third_party/MANIFEST.toml. Do not edit.\n"
"#pragma once\n\nnamespace prism {\n"
"struct ManifestPin {\n    const char* name;\n    const char* commit;\n};\n"
"inline constexpr ManifestPin kManifestPins[] = {\n${_pins}};\n}  // namespace prism\n")
configure_file("${_pins_dir}/prism/manifest_pins.hpp.tmp" "${_pins_dir}/prism/manifest_pins.hpp" COPYONLY)

# docs/TRUSTED_BASE.md (roadmap 3.2 / 8.3) and docs/VERDICTS.md (6.4) are
# written next to report.md on every run (src/prism/shipdocs.cpp,

# --- grammars/*.gbnf -> prism/ai/grammars.inc ---
# The GBNF grammars the model backends are constrained by
# (src/prism/ai/core.cpp grammar_text), embedded verbatim as raw strings.
# Generated here so the embedded copy cannot drift from grammars/.
set(_gbnf_inc "// Generated by CMake from grammars/*.gbnf. Do not edit.\n")
foreach(_g invariants harness contract explain lean_proof assumption_audit ask draft)
  set(_g_path "${PRISM_ROOT}/grammars/${_g}.gbnf")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_g_path}")
  file(READ "${_g_path}" _g_text)
  string(FIND "${_g_text}" ")GBNF\"" _g_bad)
  if(NOT _g_bad EQUAL -1)
    message(FATAL_ERROR "grammars/${_g}.gbnf contains the raw-string delimiter )GBNF\"")
  endif()
  string(TOUPPER "${_g}" _g_up)
  string(APPEND _gbnf_inc "constexpr const char* GBNF_${_g_up} = R\"GBNF(${_g_text})GBNF\";\n")
endforeach()
file(WRITE "${_pins_dir}/prism/ai/grammars.inc.tmp" "${_gbnf_inc}")
configure_file("${_pins_dir}/prism/ai/grammars.inc.tmp" "${_pins_dir}/prism/ai/grammars.inc" COPYONLY)

# --- src/prism/solver/predict_default.json -> prism/solver/predict_default.inc ---
# The built-in solver model (docs/SOLVERS.md "Learned scheduler"; written by
# `prism_ai predict --out`), embedded as raw-string pieces of at most 12000
# bytes each (MSVC caps one string literal at 16 KB); predict.cpp joins them.
set(_pm_path "${PRISM_ROOT}/src/prism/solver/predict_default.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_pm_path}")
file(READ "${_pm_path}" _pm_text)
string(STRIP "${_pm_text}" _pm_text)
string(FIND "${_pm_text}" ")JSON\"" _pm_bad)
if(NOT _pm_bad EQUAL -1)
  message(FATAL_ERROR "predict_default.json contains the raw-string delimiter )JSON\"")
endif()
string(LENGTH "${_pm_text}" _pm_len)
set(_pm_inc "// Generated by CMake from src/prism/solver/predict_default.json. Do not edit.\n")
string(APPEND _pm_inc "static const char* const kDefaultModelParts[] = {\n")
set(_pm_pos 0)
while(_pm_pos LESS _pm_len)
  string(SUBSTRING "${_pm_text}" ${_pm_pos} 12000 _pm_chunk)
  string(APPEND _pm_inc "    R\"JSON(${_pm_chunk})JSON\",\n")
  math(EXPR _pm_pos "${_pm_pos} + 12000")
endwhile()
string(APPEND _pm_inc "};\n")
file(WRITE "${_pins_dir}/prism/solver/predict_default.inc.tmp" "${_pm_inc}")
configure_file("${_pins_dir}/prism/solver/predict_default.inc.tmp" "${_pins_dir}/prism/solver/predict_default.inc" COPYONLY)

# Roadmap 2.8: the Clang-AST lints parse in process through libclang's C API.
# Only the header is needed at build time: the library is dlopen()ed at run
# time (src/prism/astlint_libclang.cpp), so a machine without libclang.so still
# runs PRISM and the layer falls back to clang as a process. Default ON when
# <clang-c/Index.h> is found (apt install libclang-18-dev).
option(PRISM_LIBCLANG "Clang-AST lints through libclang (header needed at build time)" ON)
if(PRISM_LIBCLANG)
  set(_llvm_hints)
  foreach(_v 21 20 19 18 17)
    list(APPEND _llvm_hints "/usr/lib/llvm-${_v}")
  endforeach()
  list(APPEND _llvm_hints "/usr/local/opt/llvm" "/opt/homebrew/opt/llvm" "C:/Program Files/LLVM")
  find_path(PRISM_CLANG_C_INCLUDE clang-c/Index.h
    HINTS ${_llvm_hints} PATH_SUFFIXES include)
  find_library(PRISM_LIBCLANG_LIBRARY NAMES clang libclang clang-18 clang-19 clang-20 clang-17
    HINTS ${_llvm_hints} PATH_SUFFIXES lib)
  if(PRISM_CLANG_C_INCLUDE)
    target_include_directories(prism_core SYSTEM PRIVATE "${PRISM_CLANG_C_INCLUDE}")
    target_compile_definitions(prism_core PRIVATE PRISM_HAS_LIBCLANG)
    if(PRISM_LIBCLANG_LIBRARY)
      target_compile_definitions(prism_core PRIVATE PRISM_LIBCLANG_HINT="${PRISM_LIBCLANG_LIBRARY}")
    endif()
    message(STATUS "libclang: ${PRISM_CLANG_C_INCLUDE} (run time: ${PRISM_LIBCLANG_LIBRARY}, dlopen)")
  else()
    message(STATUS "libclang header not found: the Clang-AST lints use clang as a process")
  endif()
endif()

set(_prism_cuda_lang OFF)
if(PRISM_CUDA)
  include(CheckLanguage)
  # nvcc cannot use clang++ as host. Keep CMAKE_CXX_COMPILER (clang++) for
  # prism_core; point CMAKE_CUDA_HOST_COMPILER at gcc/g++ *before*
  # check_language / enable_language so nvcc is allowed. Never pass
  # -allow-unsupported-compiler (that would silently disable the check).
  set(_prism_cuda_host "${CMAKE_CUDA_HOST_COMPILER}")
  if(NOT _prism_cuda_host)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_C_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU")
      set(_prism_cuda_host "${CMAKE_CXX_COMPILER}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang" AND UNIX)
      # Prefer gcc-12: CUDA 12.0 nvcc rejects gcc 13 unless given
      # -allow-unsupported-compiler, which would silently disable the check.
      find_program(_PRISM_CUDA_GCC NAMES gcc-12 gcc-11 gcc-10 gcc)
      if(_PRISM_CUDA_GCC)
        set(_prism_cuda_host "${_PRISM_CUDA_GCC}")
      else()
        find_program(_PRISM_CUDA_GXX NAMES g++-12 g++-11 g++-10 g++)
        if(_PRISM_CUDA_GXX)
          set(_prism_cuda_host "${_PRISM_CUDA_GXX}")
        endif()
      endif()
    endif()
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    if(_prism_cuda_host)
      set(CMAKE_CUDA_HOST_COMPILER "${_prism_cuda_host}" CACHE FILEPATH
          "Host compiler for nvcc (gcc, not clang++)" FORCE)
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR _prism_cuda_host)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
      # nvcc 12 has no C++23; do not inherit CMAKE_CXX_STANDARD 23 (clang++/prism_core).
      set(CMAKE_CUDA_STANDARD 17)
      set(CMAKE_CUDA_STANDARD_REQUIRED ON)
      enable_language(CUDA)
      set(_prism_cuda_lang ON)
      if(NOT CMAKE_CUDA_ARCHITECTURES)
        # PLAN stack is an RTX 3090 (sm_86). native needs a live GPU at configure.
        set(CMAKE_CUDA_ARCHITECTURES "86" CACHE STRING "nvcc -arch" FORCE)
      endif()
      # probsat.cu: GPU ProbSAT walkers (roadmap 3.3), mirrors src/prism/solver/cnf.cpp.
      add_library(prism_cuda SHARED src/cuda/mutate.cu src/cuda/probsat.cu)
      target_include_directories(prism_cuda PUBLIC "${PRISM_ROOT}/include")
      target_compile_features(prism_cuda PUBLIC cuda_std_17)
      set_target_properties(prism_cuda PROPERTIES
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}"
        BUILD_RPATH "$ORIGIN"
        INSTALL_RPATH "$ORIGIN")
      # prism / prism_tests / prism_gui link prism_core, not prism_native.
      # Leaving the kernel only on prism_native was a silent disable.
      target_link_libraries(prism_core PUBLIC prism_cuda)
      target_link_libraries(prism_native PUBLIC prism_cuda)
      target_compile_definitions(prism_native PUBLIC PRISM_HAS_CUDA)
      target_compile_definitions(prism_core PUBLIC PRISM_HAS_CUDA)
      message(STATUS "CUDA mutation kernel ON (nvcc host ${CMAKE_CUDA_HOST_COMPILER})")
    else()
      message(WARNING "CUDA kernel skipped (NOTRUN, not a silent disable)")
    endif()
  endif()
else()
  message(WARNING "PRISM_CUDA=OFF — CUDA kernel skipped (NOTRUN, not a silent disable)")
endif()

# --- self-fuzzing (roadmap 6.2): libFuzzer target for PRISM's own parsers ---
# cmake -DPRISM_FUZZ=ON (clang); target prism_fuzz_self; see docs/FUZZ_SELF.md.
# Built from the parser sources directly (not prism_core) so that only they
# carry the fuzzer and sanitizer instrumentation; Z3 is not linked.
option(PRISM_FUZZ "Build the libFuzzer self-fuzz target prism_fuzz_self (clang only)" OFF)
if(PRISM_FUZZ)
  include(CheckCXXSourceCompiles)
  set(CMAKE_REQUIRED_FLAGS "-fsanitize=fuzzer,address")
  check_cxx_source_compiles(
    "#include <cstddef>\n#include <cstdint>\nextern \"C\" int LLVMFuzzerTestOneInput(const uint8_t*, size_t) { return 0; }"
    PRISM_HAS_LIBFUZZER)
  unset(CMAKE_REQUIRED_FLAGS)
  if(PRISM_HAS_LIBFUZZER)
    add_executable(prism_fuzz_self
      tests/fuzz/fuzz_cparse.cpp
      src/prism/cparse.cpp
      src/prism/scope.cpp
      src/prism/laws.cpp
      src/prism/regex.cpp
      src/prism/models.cpp
      src/prism/journal.cpp
      src/prism/pir/ir_parser.cpp
      src/prism/pir/fp.cpp
      src/prism/verdict/verdict.cpp)
    target_include_directories(prism_fuzz_self PRIVATE "${PRISM_ROOT}/include" "${PRISM_ROOT}/third_party")
    set(_prism_fuzz_flags -g -O1 -fno-omit-frame-pointer
        -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=undefined)
    target_compile_options(prism_fuzz_self PRIVATE ${_prism_fuzz_flags})
    target_link_options(prism_fuzz_self PRIVATE ${_prism_fuzz_flags})
    target_link_libraries(prism_fuzz_self PRIVATE nlohmann_json)
    if(TARGET pcre2-8-static)
      target_link_libraries(prism_fuzz_self PRIVATE pcre2-8-static)
      target_compile_definitions(prism_fuzz_self PRIVATE PCRE2_STATIC)
    elseif(TARGET pcre2-8)
      target_link_libraries(prism_fuzz_self PRIVATE pcre2-8)
    endif()
    # Seed corpus (src/tools/fuzz_corpus.cpp; libFuzzer owns prism_fuzz_self's
    # main, so the corpus maker is its own small tool) and one campaign:
    #   cmake --build <dir> --target fuzz-self-run   (-DPRISM_FUZZ_SECONDS=N)
    # Fork mode keeps going after timeouts and crashes so one finding does not
    # hide the next; artifacts land in <dir>/fuzz-artifacts/.
    add_executable(prism_fuzz_corpus src/tools/fuzz_corpus.cpp)
    target_compile_definitions(prism_fuzz_corpus PRIVATE PRISM_REPO_DIR="${PRISM_ROOT}")
    set(PRISM_FUZZ_SECONDS 300 CACHE STRING "Seconds of one fuzz-self-run campaign")
    add_custom_target(fuzz-self-run
      COMMAND prism_fuzz_corpus "${CMAKE_BINARY_DIR}/fuzz-seeds"
      COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/fuzz-corpus" "${CMAKE_BINARY_DIR}/fuzz-artifacts"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_BINARY_DIR}/fuzz-seeds/libfuzzer" "${CMAKE_BINARY_DIR}/fuzz-corpus"
      COMMAND prism_fuzz_self -fork=1 -ignore_timeouts=1 -ignore_ooms=1 -ignore_crashes=1
              -max_total_time=${PRISM_FUZZ_SECONDS} -timeout=25 -rss_limit_mb=2048 -max_len=65536
              "-artifact_prefix=${CMAKE_BINARY_DIR}/fuzz-artifacts/" "${CMAKE_BINARY_DIR}/fuzz-corpus"
      COMMAND ${CMAKE_COMMAND} -E echo "fuzz-self-run: artifacts in ${CMAKE_BINARY_DIR}/fuzz-artifacts/"
      DEPENDS prism_fuzz_self prism_fuzz_corpus
      WORKING_DIRECTORY "${CMAKE_BINARY_DIR}"
      USES_TERMINAL
      VERBATIM)
  else()
    message(WARNING "PRISM_FUZZ=ON but ${CMAKE_CXX_COMPILER_ID} does not accept -fsanitize=fuzzer: "
                    "prism_fuzz_self is NOTRUN (use clang).")
    # The campaign target exists and fails loudly: NOTRUN, never a quiet pass.
    add_custom_target(fuzz-self-run
      COMMAND ${CMAKE_COMMAND} -E echo "fuzz-self-run: NOTRUN (${CMAKE_CXX_COMPILER_ID} has no -fsanitize=fuzzer; use clang)"
      COMMAND ${CMAKE_COMMAND} -E false
      VERBATIM)
  endif()
endif()
